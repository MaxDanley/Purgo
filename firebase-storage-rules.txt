// Firebase Storage Security Rules for Purgo App
// Copy these rules to your Firebase Console > Storage > Rules

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Rules for profile images
    match /profile_images/{userId}.jpg {
      // Allow read access to all authenticated users (for viewing profile pictures)
      allow read: if request.auth != null;
      
      // Allow write access only to the owner of the profile picture
      allow write: if request.auth != null 
                   && request.auth.uid == userId
                   && isValidImage();
    }
    
    // Helper function to validate image uploads
    function isValidImage() {
      return request.resource.size < 5 * 1024 * 1024  // Max 5MB
             && request.resource.contentType.matches('image/.*');  // Only images
    }
    
    // Deny access to all other paths
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}

// Instructions for applying these rules:
// 1. Go to Firebase Console (https://console.firebase.google.com/)
// 2. Select your project: purgo-1374e
// 3. Go to Storage section
// 4. Click on "Rules" tab
// 5. Replace the existing rules with the rules above
// 6. Click "Publish" to apply the changes

// What these rules do:
// - Allow any authenticated user to READ profile pictures (for viewing in friends list, leaderboard, etc.)
// - Allow users to WRITE/UPDATE only their own profile picture (userId must match auth.uid)
// - Validate that uploaded files are images and under 5MB
// - Deny all access to any other storage paths for security
